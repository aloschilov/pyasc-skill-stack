cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(direct_swiglu LANGUAGES ASC CXX)
add_executable(bench short_rows.asc)
set(CMAKE_ASC_ARCHITECTURES "dav-3510" CACHE STRING "950PR device architecture")
target_compile_options(bench PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}> $<$<COMPILE_LANGUAGE:ASC>:--enable-simt>)

add_executable(bench_vector short_rows.asc)
target_compile_definitions(bench_vector PRIVATE DIRECT_VECTOR)
target_compile_options(bench_vector PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)

# Hold register math and ownership fixed while varying DMA descriptor geometry.
foreach(layout IN ITEMS read write both)
    add_executable(bench_vector_${layout} EXCLUDE_FROM_ALL short_rows.asc)
    target_compile_definitions(bench_vector_${layout} PRIVATE DIRECT_VECTOR)
    if(layout STREQUAL "read" OR layout STREQUAL "both")
        target_compile_definitions(bench_vector_${layout} PRIVATE PACKED_FLAT_READ)
    endif()
    if(layout STREQUAL "write" OR layout STREQUAL "both")
        target_compile_definitions(bench_vector_${layout} PRIVATE PACKED_FLAT_WRITE)
    endif()
    target_compile_options(bench_vector_${layout} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)
endforeach()

foreach(order IN ITEMS serial overlap)
    add_executable(bench_vector_${order} EXCLUDE_FROM_ALL short_rows.asc)
    target_compile_definitions(bench_vector_${order} PRIVATE DIRECT_VECTOR PACKED_STAGES)
    if(order STREQUAL "serial")
        target_compile_definitions(bench_vector_${order} PRIVATE PACKED_SERIAL)
    endif()
    target_compile_options(bench_vector_${order} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)
endforeach()

# Match the generated buffer-ID protocol against native hard events. The
# overlap target supplies the InitSocState + hard-event control.
foreach(mode IN ITEMS events_pipe bufid_init bufid_pipe)
    add_executable(bench_vector_${mode} EXCLUDE_FROM_ALL short_rows.asc)
    target_compile_definitions(bench_vector_${mode} PRIVATE DIRECT_VECTOR PACKED_STAGES)
    if(mode MATCHES "^bufid")
        target_compile_definitions(bench_vector_${mode} PRIVATE STAGE_BUFID)
    endif()
    if(mode MATCHES "pipe$")
        target_compile_definitions(bench_vector_${mode} PRIVATE STAGE_TPIPE)
    endif()
    target_compile_options(bench_vector_${mode} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)
endforeach()
